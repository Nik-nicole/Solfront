🔹 Paso 1: Capturar imágenes o video con MediaPipe (detección de manos)
Vamos a usar MediaPipe para detectar manos y guardar imágenes que luego usarás para entrenar tu modelo.

🧰 Requisitos previos

📸 Código para capturar imágenes de las manos
python
Copiar
Editar
import cv2
import mediapipe as mp
import os

# Configura carpetas donde guardar imágenes
output_folder = 'dataset/seña_a'  # Puedes cambiar a seña_b, seña_c, etc.
os.makedirs(output_folder, exist_ok=True)

# Inicializa MediaPipe Hands
mp_hands = mp.solutions.hands
hands = mp_hands.Hands()
mp_drawing = mp.solutions.drawing_utils

cap = cv2.VideoCapture(0)
img_count = 0

while cap.isOpened():
    success, frame = cap.read()
    if not success:
        break

    # Convertir color
    image = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
    results = hands.process(image)

    if results.multi_hand_landmarks:
        for hand_landmarks in results.multi_hand_landmarks:
            # Dibuja los puntos de la mano
            mp_drawing.draw_landmarks(frame, hand_landmarks, mp_hands.HAND_CONNECTIONS)
            
            # Guarda imagen si se detecta una mano
            img_path = os.path.join(output_folder, f'{img_count}.jpg')
            cv2.imwrite(img_path, frame)
            img_count += 1
            print(f'Guardada: {img_path}')

    cv2.imshow('Detección de manos', frame)
    if cv2.waitKey(1) & 0xFF == 27 or img_count >= 100:  # ESC para salir o 100 imágenes
        break

cap.release()
cv2.destroyAllWindows()


🔹 Paso 2: Entrenar modelo en Keras con las imágenes
Aquí un ejemplo sencillo para clasificar dos señas: A y B.


🧠 Código base para entrenamiento

python
Copiar
Editar
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Conv2D, MaxPooling2D, Flatten, Dense
from tensorflow.keras.preprocessing.image import ImageDataGenerator

train_datagen = ImageDataGenerator(rescale=1./255)

train_data = train_datagen.flow_from_directory(
    'dataset/',  # Debe tener subcarpetas como "seña_a", "seña_b"
    target_size=(128, 128),
    batch_size=32,
    class_mode='categorical'
)

model = Sequential([
    Conv2D(32, (3, 3), activation='relu', input_shape=(128, 128, 3)),
    MaxPooling2D(2, 2),
    Flatten(),
    Dense(128, activation='relu'),
    Dense(2, activation='softmax')  # Cambia según número de señas
])

model.compile(optimizer='adam', loss='categorical_crossentropy', metrics=['accuracy'])
model.fit(train_data, epochs=10)

# Guarda el modelo en formato Keras
model.save('modelo_señas.h5')


🔹 Paso 3: Convertir modelo Keras a .tflite

🔄 Conversión a .tflite
python
Copiar
Editar
import tensorflow as tf

# Carga modelo
model = tf.keras.models.load_model('modelo_señas.h5')

# Convertir a TFLite
converter = tf.lite.TFLiteConverter.from_keras_model(model)
tflite_model = converter.convert()

# Guardar archivo .tflite
with open('modelo_señas.tflite', 'wb') as f:
    f.write(tflite_model)

print("Modelo convertido y guardado como 'modelo_señas.tflite'")
